// 업체명 완벽 정제 함수 function sanitizeCompany(raw) { if (!raw) return ''; // 1. 법인명 표기 제거 ((주), 주식회사, (유), 유한회사 등) let name = raw.replace(/\((?:주|유|합|사|재)\)|주식회사|유한회사|주\.|㈜|㈲/g, ' '); // 2. 시험 항목 / 보고서 비고 / 데이터 관련 키워드 const stopWords = [ '피부', '탄력', '개선', '개선율', '보습', '주름', '미백', '진정', '모공', '각질', '피지', '외', '결과데이터', '결과', '데이터', 'lab', 'raw', 'data' ]; // 단어 단위 검사 (비고 단어가 시작되면 그 뒤는 전부 버림) let tokens = name.trim().split(/[\s_]+/); let cleanTokens = []; for (let token of tokens) { let lower = token.toLowerCase(); let hasStopWord = stopWords.some(sw => lower.includes(sw)); if (hasStopWord) { break; } cleanTokens.push(token); } name = cleanTokens.join(' '); // 3. 영문자 및 잔여 특수기호 제거 name = name.replace(/[a-zA-Z]+/g, ' '); name = name.replace(/[\+\-_/\\&~*"]/g, ' '); return name.replace(/\s{2,}/g, ' ').trim(); } // 제외할 더미 텍스트 목록 (이 텍스트만 단독으로 남은 행은 아예 출력 제외) const excludeList = ['결과데이터', '데이터', '결과', 'rawdata', 'lab']; function isInvalidCompany(name) { if (!name) return true; let compact = name.replace(/\s+/g, '').toLowerCase(); return excludeList.includes(compact); }
